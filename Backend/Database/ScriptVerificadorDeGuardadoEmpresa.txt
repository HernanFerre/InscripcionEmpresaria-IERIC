/* VERIFICAR EMPRESA COMÚN O UTE */

USE [SolicitudInscripcionEmpresariaDigital];
GO

DECLARE @Cuit BIGINT = 30123456781;

DECLARE @EmpresaId BIGINT =
(
    SELECT E.Id
    FROM dbo.Empresa AS E
    WHERE E.Cuit = @Cuit
);

/* EMPRESA PRINCIPAL Y POSIBLES INTEGRANTES */
SELECT
    'dbo.Empresa' AS TablaConsultada,
    E.*
FROM dbo.Empresa AS E
WHERE E.Id = @EmpresaId
   OR E.Id IN
   (
       SELECT EC.IdEmpresaIntegrante
       FROM dbo.EmpresasCuit AS EC
       WHERE EC.IdEmpresa = @EmpresaId
   )
ORDER BY
    CASE WHEN E.Id = @EmpresaId THEN 0 ELSE 1 END,
    E.LegacyId,
    E.Id;

/* SOLICITUD VINCULADA */
SELECT
    'dbo.SolicitudInscripcionDigitalEmpresaria'
        AS TablaConsultada,
    S.*
FROM dbo.SolicitudInscripcionDigitalEmpresaria AS S
WHERE S.Idempresa = @EmpresaId
ORDER BY S.Id DESC;

/* RELACIONES DE LA UTE */
SELECT
    'dbo.EmpresasCuit' AS TablaConsultada,
    EC.*
FROM dbo.EmpresasCuit AS EC
WHERE EC.IdEmpresa = @EmpresaId
ORDER BY EC.IdEmpresaIntegrante;

/* LIMPIEZA COMPLETA DE EMPRESAS Y SOLICITUDES */

USE [SolicitudInscripcionEmpresariaDigital];
GO

SET XACT_ABORT ON;

BEGIN TRY
    BEGIN TRANSACTION;

    /* 1. Relaciones entre empresas */
    DELETE FROM dbo.EmpresasCuit;

    /* 2. Solicitudes */
    DELETE FROM dbo.SolicitudInscripcionDigitalEmpresaria;

    /* 3. Empresas principales e integrantes */
    DELETE FROM dbo.Empresa;

    /* Reiniciar los identificadores */
    DBCC CHECKIDENT
    (
        'dbo.SolicitudInscripcionDigitalEmpresaria',
        RESEED,
        0
    ) WITH NO_INFOMSGS;

    DBCC CHECKIDENT
    (
        'dbo.Empresa',
        RESEED,
        0
    ) WITH NO_INFOMSGS;

    COMMIT TRANSACTION;
END TRY
BEGIN CATCH
    IF @@TRANCOUNT > 0
        ROLLBACK TRANSACTION;

    THROW;
END CATCH;

/* VERIFICACIÓN: LAS TRES DEBEN MOSTRAR 0 */
SELECT
    'dbo.Empresa' AS TablaConsultada,
    COUNT(*) AS RegistrosRestantes
FROM dbo.Empresa

UNION ALL

SELECT
    'dbo.SolicitudInscripcionDigitalEmpresaria',
    COUNT(*)
FROM dbo.SolicitudInscripcionDigitalEmpresaria

UNION ALL

SELECT
    'dbo.EmpresasCuit',
    COUNT(*)
FROM dbo.EmpresasCuit;
